```typescript
export const prerender = false;

import type { APIRoute } from 'astro';
import { Hono } from 'hono';
import { zValidator } from '@hono/zod-validator';
import { drizzle } from 'drizzle-orm/d1';
import { eq, count, lte } from 'drizzle-orm';
import { Resend } from 'resend';
import { waitlist } from '../../db/schema';
import { waitlistSchema } from '../../lib/validation';
import { generateReferralCode } from '../../lib/referral';
import WelcomeEmail from '../../emails/WelcomeEmail';

type Bindings = {
  DB: D1Database;
  RESEND_API_KEY: string;
};

const app = new Hono<{ Bindings: Bindings }>();

app.post('/', zValidator('form', waitlistSchema), async (c) => {
  const { email, name, source, referredBy } = c.req.valid('form');
  const db = drizzle(c.env.DB);

  // Check for duplicate email
  const existing = await db.select()
    .from(waitlist)
    .where(eq(waitlist.email, email.toLowerCase()))
    .limit(1);

  if (existing.length > 0) {
    return c.json({
      error: 'This email is already on the waitlist!',
      position: null,
      referralCode: existing[0].referralCode,
    }, 409);
  }

  // Generate unique referral code
  let referralCode = generateReferralCode();
  let attempts = 0;
  while (attempts < 5) {
    const codeExists = await db.select()
      .from(waitlist)
      .where(eq(waitlist.referralCode, referralCode))
      .limit(1);
    if (codeExists.length === 0) break;
    referralCode = generateReferralCode();
    attempts++;
  }

  // Insert new entry
  const [entry] = await db.insert(waitlist).values({
    email: email.toLowerCase(),
    name: name || null,
    source: source || null,
    referralCode,
    referredBy: referredBy || null,
    referralCount: 0,
    createdAt: new Date().toISOString(),
  }).returning();

  // Update referrer's count if valid referral
  if (referredBy) {
    await db.update(waitlist)
      .set({
        referralCount: db.$dynamic()
          .increment(waitlist.referralCount, 1)
      })
      .where(eq(waitlist.referralCode, referredBy));
    // Note: Drizzle D1 increment syntax may vary - fallback to raw SQL if needed:
    // await db.run(sql`UPDATE waitlist SET referral_count = referral_count + 1 WHERE referral_code = ${referredBy}`);
  }

  // Get position (count of entries with id <= this entry's id)
  const [positionResult] = await db.select({ count: count() })
    .from(waitlist)
    .where(lte(waitlist.id, entry.id));

  const position = positionResult.count;

  // Send confirmation email
  try {
    const resend = new Resend(c.env.RESEND_API_KEY);
    await resend.emails.send({
      from: 'Illoominate <hello@illoominate.com>',
      to: email,
      subject: `You're #${position} on the Illoominate waitlist!`,
      react: WelcomeEmail({ name, position, referralCode }),
    });
  } catch (emailError) {
    // Log but don't fail the request - user is still on waitlist
    console.error('Failed to send email:', emailError);
  }

  return c.json({
    success: true,
    position,
    referralCode,
  });
});

// Export for Astro
export const POST: APIRoute = async (context) => {
  return app.fetch(context.request, {
    DB: context.locals.runtime.env.DB,
    RESEND_API_KEY: context.locals.runtime.env.RESEND_API_KEY,
  });
};
```

**Important notes:**
- Use `export const prerender = false;` - API routes must be SSR
- Hono handles validation via zValidator middleware
- Email failures are logged but don't fail the request
- Duplicate emails return 409 with existing referral code
- Referral code collision is handled with retry loop
