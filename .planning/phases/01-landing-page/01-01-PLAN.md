<task type="auto">
  <name>Task 2: Set up D1 database with Drizzle schema</name>
  <files>
    src/db/schema.ts
    src/db/index.ts
    drizzle.config.ts
    wrangler.jsonc
  </files>
  <action>
    1. Create D1 database:
       ```bash
       npx wrangler d1 create illoominate-waitlist
       ```
       Copy the database_id from output and update wrangler.jsonc.

    2. Create `src/db/schema.ts` with Drizzle schema:
       ```typescript
       import { sqliteTable, text, integer } from 'drizzle-orm/sqlite-core';

       export const waitlist = sqliteTable('waitlist', {
         id: integer('id').primaryKey({ autoIncrement: true }),
         email: text('email').notNull().unique(),
         name: text('name'),
         source: text('source'),
         referralCode: text('referral_code').notNull().unique(),
         referredBy: text('referred_by'),
         referralCount: integer('referral_count').default(0),
         createdAt: text('created_at').notNull(),
       });

       export const quizResponses = sqliteTable('quiz_responses', {
         id: integer('id').primaryKey({ autoIncrement: true }),
         waitlistId: integer('waitlist_id').references(() => waitlist.id),
         platform: text('platform'),
         teamSize: text('team_size'),
         painPoints: text('pain_points'),
         createdAt: text('created_at').notNull(),
       });
       ```

    3. Create `src/db/index.ts` for database client helper:
       ```typescript
       import { drizzle } from 'drizzle-orm/d1';
       import * as schema from './schema';

       export function getDb(d1: D1Database) {
         return drizzle(d1, { schema });
       }

       export * from './schema';
       ```

    4. Create `drizzle.config.ts`:
       ```typescript
       import { defineConfig } from 'drizzle-kit';

       export default defineConfig({
         schema: './src/db/schema.ts',
         out: './drizzle',
         dialect: 'sqlite',
         driver: 'd1-http',
       });
       ```

    5. Apply schema to D1:
       ```bash
       npx drizzle-kit generate
       npx wrangler d1 migrations apply illoominate-waitlist --local
       ```
  </action>
  <verify>
    - `npx wrangler d1 execute illoominate-waitlist --local --command "SELECT name FROM sqlite_master WHERE type='table'"` shows waitlist and quiz_responses tables
    - `npx drizzle-kit check` passes
  </verify>
  <done>
    D1 database created, Drizzle schema defined, migrations applied locally.
  </done>
</task>
