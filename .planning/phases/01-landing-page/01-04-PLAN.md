```typescript
export const prerender = false;

import type { APIRoute } from 'astro';
import { drizzle } from 'drizzle-orm/d1';
import { quizResponses } from '../../db/schema';
import { quizSchema } from '../../lib/validation';

export const POST: APIRoute = async (context) => {
  const db = drizzle(context.locals.runtime.env.DB);

  try {
    const body = await context.request.json();
    const data = quizSchema.parse(body);

    await db.insert(quizResponses).values({
      waitlistId: data.waitlistId,
      platform: data.platform || null,
      teamSize: data.teamSize || null,
      painPoints: data.painPoints ? JSON.stringify(data.painPoints) : null,
      createdAt: new Date().toISOString(),
    });

    return new Response(JSON.stringify({ success: true }), {
      status: 200,
      headers: { 'Content-Type': 'application/json' },
    });
  } catch (error) {
    return new Response(JSON.stringify({ error: 'Invalid data' }), {
      status: 400,
      headers: { 'Content-Type': 'application/json' },
    });
  }
};
```

```typescript
import { defineMiddleware } from 'astro:middleware';

export const onRequest = defineMiddleware(async (context, next) => {
  const url = new URL(context.request.url);

  // Protect admin routes
  if (url.pathname.startsWith('/admin') || url.pathname.startsWith('/api/admin')) {
    const authHeader = context.request.headers.get('Authorization');
    const token = authHeader?.replace('Bearer ', '');

    // Also check query param for simple browser access
    const queryToken = url.searchParams.get('token');

    const validToken = context.locals.runtime?.env?.ADMIN_TOKEN;

    if (!validToken || (token !== validToken && queryToken !== validToken)) {
      return new Response('Unauthorized', { status: 401 });
    }
  }

  return next();
});
```

Create src/pages/api/admin/waitlist.ts:
```typescript
export const prerender = false;

import type { APIRoute } from 'astro';
import { drizzle } from 'drizzle-orm/d1';
import { desc, like, count } from 'drizzle-orm';
import { waitlist, quizResponses } from '../../../db/schema';

export const GET: APIRoute = async (context) => {
  const db = drizzle(context.locals.runtime.env.DB);
  const url = new URL(context.request.url);

  const format = url.searchParams.get('format') || 'json';
  const search = url.searchParams.get('search') || '';
  const page = parseInt(url.searchParams.get('page') || '1');
  const limit = parseInt(url.searchParams.get('limit') || '50');
  const offset = (page - 1) * limit;

  // Build query
  let query = db.select({
    id: waitlist.id,
    email: waitlist.email,
    name: waitlist.name,
    source: waitlist.source,
    referralCode: waitlist.referralCode,
    referredBy: waitlist.referredBy,
    referralCount: waitlist.referralCount,
    createdAt: waitlist.createdAt,
  }).from(waitlist);

  if (search) {
    query = query.where(like(waitlist.email, `%${search}%`));
  }

  const entries = await query
    .orderBy(desc(waitlist.createdAt))
    .limit(limit)
    .offset(offset);

  const [totalResult] = await db.select({ count: count() }).from(waitlist);
  const total = totalResult.count;

  // CSV export
  if (format === 'csv') {
    const headers = ['id', 'email', 'name', 'source', 'referral_code', 'referred_by', 'referral_count', 'created_at'];
    const rows = entries.map(e => [
      e.id,
      e.email,
      e.name || '',
      e.source || '',
      e.referralCode,
      e.referredBy || '',
      e.referralCount,
      e.createdAt,
    ]);

    const csv = [headers.join(','), ...rows.map(r => r.map(v => `"${v}"`).join(','))].join('\n');

    return new Response(csv, {
      headers: {
        'Content-Type': 'text/csv',
        'Content-Disposition': `attachment; filename="waitlist-${new Date().toISOString().split('T')[0]}.csv"`,
      },
    });
  }

  return new Response(JSON.stringify({
    entries,
    pagination: {
      page,
      limit,
      total,
      pages: Math.ceil(total / limit),
    },
  }), {
    headers: { 'Content-Type': 'application/json' },
  });
};
```

```astro
  <script>
    // Get token from URL (simple auth for admin)
    const token = new URLSearchParams(window.location.search).get('token') || '';
    let currentPage = 1;
    let searchQuery = '';

    async function fetchWaitlist(page = 1, search = '') {
      const params = new URLSearchParams({
        page: page.toString(),
        limit: '50',
        ...(search && { search }),
      });

      const res = await fetch(`/api/admin/waitlist?${params}`, {
        headers: { 'Authorization': `Bearer ${token}` },
      });

      if (!res.ok) {
        alert('Unauthorized or error fetching data');
        return;

      const data = await res.json();
      renderTable(data.entries);
      renderPagination(data.pagination);
      renderStats(data.pagination.total);

      const tbody = document.getElementById('waitlist-table');
      tbody.innerHTML = entries.map((e, i) => `

          <td class="px-4 py-3 text-sm text-gray-400">${e.source || '-'}</td>
          <td class="px-4 py-3 text-sm text-primary-400">${e.referralCount}</td>
          <td class="px-4 py-3 text-sm text-gray-500">${new Date(e.createdAt).toLocaleDateString()}</td>
        </tr>
      `).join('');
    }

    function renderPagination(pagination) {
      const div = document.getElementById('pagination');
      const buttons = [];
      for (let i = 1; i <= pagination.pages; i++) {
        buttons.push(`
          <button class="px-3 py-1 rounded ${i === pagination.page ? 'bg-primary-600' : 'bg-gray-800 hover:bg-gray-700'}"
                  onclick="goToPage(${i})">${i}</button>
        `);
      }
      div.innerHTML = buttons.join('');
    }

    function renderStats(total) {
      document.getElementById('stats').innerHTML = `
        <div class="bg-gray-900 rounded-xl p-6">
          <div class="text-3xl font-bold text-white">${total}</div>
          <div class="text-sm text-gray-400">Total signups</div>
        </div>
      `;
    }

      currentPage = page;
      fetchWaitlist(page, searchQuery);

      fetchWaitlist(currentPage, searchQuery);

        fetchWaitlist(1, searchQuery);
      }, 300);
    });

    fetchWaitlist();
```
