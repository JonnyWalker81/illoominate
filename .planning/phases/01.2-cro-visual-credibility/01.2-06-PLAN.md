       Update the showStep function in JavaScript to show typing indicator:
       ```typescript
       function showStep(step: number) {
         const currentStepEl = document.querySelector(`[data-step="${currentStep}"]`) as HTMLElement;
         const nextStepEl = document.querySelector(`[data-step="${step}"]`) as HTMLElement;

         if (currentStepEl && nextStepEl && currentStep !== step) {
           currentStepEl.classList.add('exiting');

           // Show typing indicator briefly
           const typingEl = document.getElementById('typing-indicator');
           if (typingEl && step > currentStep) {
             typingEl.classList.remove('hidden');
           }

           setTimeout(() => {
             currentStepEl.classList.add('hidden');
             currentStepEl.classList.remove('exiting');

             // Hide typing indicator
             if (typingEl) {
               typingEl.classList.add('hidden');
             }

             nextStepEl.classList.remove('hidden');
             currentStep = step;
           }, 400); // Slightly longer to show typing
         }

         updateProgress(step);
       }
       ```
